import torch
import torch.nn as nn

class CNN(nn.Module):
    def __init__(self):
        super().__init__() 
        
        self.conv1 = nn.Conv2d(
            in_channels=3, 
            out_channels=6, 
            kernel_size=3, 
            stride=1, 
            padding=0
        )
        
        self.maxpool1 = nn.MaxPool2d(
            kernel_size=3,  # 池化窗口大小：3×3
            ceil_mode=True  # 向上取整模式，保留不完整的边缘窗口
            # stride默认等于kernel_size=3，也可以显式写stride=3，效果完全一致
        )

    def forward(self, x):
        # 前向传播：**先卷积提取特征，再池化下采样**（CNN标准顺序）
        # x: [b, 3, h, w]
        x = self.conv1(x)       # 第一步：过卷积层，生成6通道特征图, [b, 6, h', w']
        # h'=(h+2*p-k)/s+1
        x = self.maxpool1(x)    # 第二步：过池化层，对特征图下采样
        return x

class CNN(nn.Module):
    def __init__(self):
        super().__init__()
        self.conv1 = nn.Conv2d(3,6,3)
        self.pool = nn.MaxPool2d(2)
        self.flatten = nn.Flatten()
        
        # 输入是 32*32，这个是对的！
        self.fc = nn.Linear(6*5*5, 10)

    def forward(self, x):
        # 假设输入是 [B,3,32,32]
        x = self.conv1(x)   # [B,6,30,30]  ✅ 你写对了
        x = self.pool(x)    # [B,6,15,15]    ✅ 你写对了
        x = self.flatten(x) # [B,6*15*15]    ✅
        x = self.fc(x)      # [B,10]
        return x
# 训练
model = CNN()

    loss = criterion(outputs, labels)  # 计算损失
    loss.backward()

# 1. 单词词典：词 → 数字编号
vocab = {"I":1, "love":2, "AI":3, "ML":4}

# 2. 真实句子 → 转成数字
sentences = [[1,2,3], [1,2,4]]  # 2句话，每句3个词
input = torch.tensor(sentences)

# 3. 定义Embedding：词典共5个词(0-4)，每个词转4维向量
emb = nn.Embedding(num_embeddings=5, embedding_dim=4)

# 4. 查表：数字 → 向量
output = emb(input)

# 结果
print("输入形状:", input.shape)  # (2, 3)
print("输出形状:", output.shape)# (2, 3, 4)

import torch
import torch.nn as nn

class MyLayer(nn.Module):
    def __init__(self):
        super().__init__()
        # ✅ 正确：可训练参数
        self.weight = nn.Parameter(torch.randn(5, 3))  
        # ❌ 错误：普通张量，不参与训练
        self.bias = torch.randn(5)  

model = MyLayer()
# 只会打印上面的 weight，bias 不会出现
print(list(model.parameters()))

VisualEncoder

离线特征

  1. 用 SigLIP2 自带的 preprocess 把图片 resize / crop / normalize
  2. 输入分辨率 256，patch size 16。所以一张图会被切成 16 × 16 = 256 个 patch
  3. 每个 patch 变成向量
  4. 过 Transformer trunk, 得到[B,256,1152]，也就是一张图片有256个patch，每个patch有1152维

class SigLIP2AttentionPool(nn.Module):
    """AttentionPoolLatent matching SigLIP2's pretrained structure.

    This replicates the exact architecture from:
    model.visual.trunk.attn_pool in open_clip SigLIP2

    Also includes the trunk.norm that should be applied before attn_pool.

    Can be initialized from pretrained weights via load_from_siglip2().
    """

    def __init__(self, dim: int = 1152, mlp_ratio: float = 4304 / 1152):
        super().__init__()
        self.dim = dim
        mlp_hidden = int(dim * mlp_ratio)

        # Learnable query (latent)
        self.latent = nn.Parameter(torch.randn(1, 1, dim) * 0.02)

        # Attention components (matching SigLIP2 structure)
        self.q = nn.Linear(dim, dim, bias=True)
        self.kv = nn.Linear(dim, 2 * dim, bias=True)
        self.q_norm = nn.Identity()
        self.k_norm = nn.Identity()
        self.proj = nn.Linear(dim, dim, bias=True)
        self.proj_drop = nn.Dropout(p=0.0)

        # Post-attention norm and MLP (matching SigLIP2 structure)
        self.norm = nn.LayerNorm(dim, eps=1e-6)
        self.mlp = nn.Sequential(
            nn.Linear(dim, mlp_hidden, bias=True),  # fc1
            nn.GELU(),  # GELUTanh in SigLIP2, but GELU is close enough
            nn.Dropout(p=0.0),  # drop1
            nn.Linear(mlp_hidden, dim, bias=True),  # fc2
            nn.Dropout(p=0.0),  # drop2
        )

        self.num_heads = 16  # SigLIP2 uses 16 heads for 1152 dim
        self.head_dim = dim // self.num_heads
        self.scale = self.head_dim**-0.5

    def load_from_siglip2(self, siglip2_attn_pool, siglip2_trunk_norm=None):
        """Load weights from pretrained SigLIP2 attn_pool module.

        Args:
            siglip2_attn_pool: The attn_pool module from model.visual.trunk.attn_pool
            siglip2_trunk_norm: The norm module from model.visual.trunk.norm (optional)
        """
        with torch.no_grad():
            # Copy attention weights
            self.q.weight.copy_(siglip2_attn_pool.q.weight)
            self.q.bias.copy_(siglip2_attn_pool.q.bias)
            self.kv.weight.copy_(siglip2_attn_pool.kv.weight)
            self.kv.bias.copy_(siglip2_attn_pool.kv.bias)
            self.proj.weight.copy_(siglip2_attn_pool.proj.weight)
            self.proj.bias.copy_(siglip2_attn_pool.proj.bias)

            # Copy post-attention norm weights
            self.norm.weight.copy_(siglip2_attn_pool.norm.weight)
            self.norm.bias.copy_(siglip2_attn_pool.norm.bias)

            # Copy MLP weights
            self.mlp[0].weight.copy_(siglip2_attn_pool.mlp.fc1.weight)
            self.mlp[0].bias.copy_(siglip2_attn_pool.mlp.fc1.bias)
            self.mlp[3].weight.copy_(siglip2_attn_pool.mlp.fc2.weight)
            self.mlp[3].bias.copy_(siglip2_attn_pool.mlp.fc2.bias)

            # Copy latent query if available
            if hasattr(siglip2_attn_pool, "latent"):
                self.latent.copy_(siglip2_attn_pool.latent)

        print("SigLIP2AttentionPool: loaded pretrained weights from SigLIP2")

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """
        Args:
            x: Visual tokens [B, N, C] (patch tokens from SigLIP2 backbone)
               Note: These should be UNNORMALIZED tokens (before trunk.norm)

        Returns:
            Pooled semantic embedding [B, C] matching SigLIP2 CLS output
        """
        B, N, C = x.shape

        # Expand latent query for batch
        q = self.latent.expand(B, -1, -1)  # [B, 1, C]

        # Compute Q, K, V
        q = self.q(q)  # [B, 1, C]
        kv = self.kv(x)  # [B, N, 2*C]
        k, v = kv.chunk(2, dim=-1)  # [B, N, C] each

        # Apply norms (Identity in SigLIP2)
        q = self.q_norm(q)
        k = self.k_norm(k)

        # Reshape for multi-head attention
        q = q.view(B, 1, self.num_heads, self.head_dim).transpose(1, 2)  # [B, H, 1, D]
        k = k.view(B, N, self.num_heads, self.head_dim).transpose(1, 2)  # [B, H, N, D]
        v = v.view(B, N, self.num_heads, self.head_dim).transpose(1, 2)  # [B, H, N, D]

        # Attention
        attn = (q @ k.transpose(-2, -1)) * self.scale  # [B, H, 1, N]
        attn = attn.softmax(dim=-1)

        # Apply attention to values
        out = (attn @ v).transpose(1, 2).contiguous().view(B, 1, C)  # [B, 1, C]

        # Project and add residual
        out = self.proj(out)
        out = self.proj_drop(out)

        # MLP with residual
        out = out + self.mlp(self.norm(out))
        out = out / out.norm(dim=-1, keepdim=True)

        return out.squeeze(1)  # [B, C]

class LowRankSemanticAdaptorProj(nn.Module):
    """Low-rank adaptor with optional projection for aligned semantic embeddings.

    For already-aligned embeddings (e.g., SigLIP2 visual/text), we use:
    1. Low-rank residual adaptation to preserve alignment while allowing fine-tuning
    2. Optional projection to change output dimension

    Design: (Identity + low-rank residual) -> optional projection
        adapted = x + scale * (down_proj -> GELU -> up_proj)(x)
        output = proj(adapted) if out_dim != dim else adapted

    Key features:
    - Very low rank (e.g., 16) to minimize parameters and prevent overfitting
    - Residual connection preserves original alignment by default
    - Learnable scale starts small (0.1) for gradual adaptation
    - Optional projection when output dim differs from input dim

    Args:
        dim: Input dimension (e.g., 1152 for SigLIP2)
        out_dim: Output dimension (default None = same as dim, no projection)
        rank: Bottleneck rank for adaptation (default 16)
        init_scale: Initial scale for residual (default 0.1)
    """

    def __init__(
        self,
        dim: int,
        out_dim: int = None,
        rank: int = 16,
        init_scale: float = 0.1,
    ):
        super().__init__()
        self.dim = dim
        self.out_dim = out_dim if out_dim is not None else dim
        self.rank = rank
        self.use_projection = self.out_dim != dim

        # Low-rank bottleneck: dim -> rank -> dim
        self.down_proj = nn.Linear(dim, rank, bias=False)
        self.up_proj = nn.Linear(rank, dim, bias=False)

        # Initialize for near-identity at start
        nn.init.kaiming_normal_(
            self.down_proj.weight, mode="fan_in", nonlinearity="linear"
        )
        nn.init.zeros_(self.up_proj.weight)

        # Learnable scale for residual (starts small)
        self.scale = nn.Parameter(torch.tensor(init_scale))

        # Optional projection if out_dim != dim
        if self.use_projection:
            self.proj = nn.Sequential(
                nn.LayerNorm(dim),
                nn.Linear(dim, self.out_dim),
            )
        else:
            self.proj = nn.Identity()

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        """Forward: (x + scale * low_rank_residual(x)) -> optional projection.

        Args:
            x: Input tensor [B, dim]

        Returns:
            Adapted tensor [B, out_dim]
        """
        # Low-rank residual adaptation
        residual = self.up_proj(F.gelu(self.down_proj(x)))
        adapted = x + self.scale * residual

        # Optional projection
        return self.proj(adapted)

    def extra_repr(self) -> str:
        return f"dim={self.dim}, out_dim={self.out_dim}, rank={self.rank}, scale={self.scale.item():.4f}"

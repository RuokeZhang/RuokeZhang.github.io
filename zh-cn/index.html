# image_encoder - ResNet or Vision Transformer
# text_encoder - CBOW or Text Transformer
# I[n, h, w, c] - minibatch of aligned images
# T[n, l] - minibatch of aligned texts
# W_i[d_i, d_e] - learned proj of image to embed
# W_t[d_t, d_e] - learned proj of text to embed
# t - learned temperature parameter
# extract feature representations of each modality
I_f = image_encoder(I) #[n, d_i]
T_f = text_encoder(T) #[n, d_t]
# joint multimodal embedding [n, d_e]
I_e = l2_normalize(np.dot(I_f, W_i), axis=1) #[n, d_e]
T_e = l2_normalize(np.dot(T_f, W_t), axis=1) #[n, n]
# scaled pairwise cosine similarities [n, n]
logits = np.dot(I_e, T_e.T) * np.exp(t)
# symmetric loss function
labels = np.arange(n)
loss_i = cross_entropy_loss(logits, labels, axis=1) # image-to-text loss
loss_t = cross_entropy_loss(logits, labels, axis=0) 
loss = (loss_i + loss_t)/2

class VisionTransformer(nn.Module):
    def __init__(self, input_resolution: int, patch_size: int, width: int, layers: int, heads: int, output_dim: int):
        super().__init__()
        self.input_resolution = input_resolution
        self.output_dim = output_dim
        self.conv1 = nn.Conv2d(in_channels=3, out_channels=width, kernel_size=patch_size, stride=patch_size, bias=False) #width表示Transformer 里每个 token 的隐藏维度。
        # grid=input_resolution/patch_size
        # 经过conv1变成[B, width, grid, grid]
        scale = width ** -0.5
        self.class_embedding = nn.Parameter(scale * torch.randn(width)) #class token, 一个专门负责“收集整张图信息”的特殊 token
        self.positional_embedding = nn.Parameter(scale * torch.randn((input_resolution // patch_size) ** 2 + 1, width)) # [(input_resolution//patch_size)2+1, width]
        self.ln_pre = LayerNorm(width)

        self.transformer = Transformer(width, layers, heads)

        self.ln_post = LayerNorm(width)
        self.proj = nn.Parameter(scale * torch.randn(width, output_dim))

    def forward(self, x: torch.Tensor):
        x = self.conv1(x)  # shape = [*, width, grid, grid]
        x = x.reshape(x.shape[0], x.shape[1], -1)  # shape = [*, width, grid ** 2]
        x = x.permute(0, 2, 1)  # shape = [*, grid ** 2, width]
        x = torch.cat([self.class_embedding.to(x.dtype) + torch.zeros(x.shape[0], 1, x.shape[-1], dtype=x.dtype, device=x.device), x], dim=1)  # shape = [*, grid ** 2 + 1, width]
        x = x + self.positional_embedding.to(x.dtype)
        x = self.ln_pre(x)

        x = x.permute(1, 0, 2)  # NLD -> LND
        x = self.transformer(x)
        x = x.permute(1, 0, 2)  # LND -> NLD

        x = self.ln_post(x[:, 0, :])

        if self.proj is not None:
            x = x @ self.proj

        return x

import torch
import torch.nn as nn

class CNN(nn.Module):
    def __init__(self):
        super().__init__() 
        
        self.conv1 = nn.Conv2d(
            in_channels=3, 
            out_channels=6, 
            kernel_size=3, 
            stride=1, 
            padding=0
        )
        
        self.maxpool1 = nn.MaxPool2d(
            kernel_size=3,  # 池化窗口大小：3×3
            ceil_mode=True  # 向上取整模式，保留不完整的边缘窗口
            # stride默认等于kernel_size=3，也可以显式写stride=3，效果完全一致
        )

    def forward(self, x):
        # 前向传播：**先卷积提取特征，再池化下采样**（CNN标准顺序）
        # x: [b, 3, h, w]
        x = self.conv1(x)       # 第一步：过卷积层，生成6通道特征图, [b, 6, h', w']
        # h'=(h+2*p-k)/s+1
        x = self.maxpool1(x)    # 第二步：过池化层，对特征图下采样
        return x

Dynamo

virtual nodes

  • compared to basic consistent hashing, each node is assigned to multiple points on the ring

  • advantages

    • If a node becomes unavailable, the load handled by this node is evenly dispersed across the remaining available nodes.
    • When a new node is added to the system, the new node accepts a roughly equivalent amount of load from each of the other available nodes.
    • The number of virtual nodes that a node is responsible for can be decided based on its capacity, accounting for heterogeneity in the physical infrastructure.

  • solved consistent hashing’s problems

File System

发表于： 2026-01-05 分类于： Operating System

文件系统

inode

每个文件都有唯一的 inode, 只在当前文件系统唯一

存放的信息有

  • 文件权限（File Permissions）：读（r=read）、写（w=write）、执行（x=execute），示例：rw-r–r–（所有者可读可写，所属组和其他用户仅可读）
  • 文件所有者和所属组（File Owner & Group）：
    • 所有者 UID（User ID）
    • 所属组 GID（Group ID）
  • 文件大小（File Size）：单位字节（Byte），区分普通文件（存储内容大小）和目录文件（存储目录项的大小）
  • 时间戳（Timestamps）：3 个核心时间属性
    • ctime（Change Time）：inode 元数据修改时间（如权限、所有者变更）
    • mtime（Modify Time）：文件内容修改时间
    • atime（Access Time）：文件内容最后访问时间
  • 数据块指针（Data Block Pointers）：最核心字段，直接或间接指向存储文件内容的数据块物理地址
  • 链接数（Link Count）：指向该 inode 的硬链接数量，默认新建文件时 link count = 1

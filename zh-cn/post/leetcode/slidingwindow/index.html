def minSubArrayLen(self, target: int, nums: List[int]) -> int:
    s,l=0,0
    res=inf
    for r, x in enumerate(nums):
        s+=x
        while s>=target:
            res=min(res, r-l+1)
            s-=nums[l]
            l+=1
    return res if res<inf else 0

def soln(password: str) -> int:
    count = 1
    seen = set()

    for i in range(len(password)):
        c = password[i]
        if c in seen:
            seen = {c}
            count += 1
        else:
            seen.add(c)

    print(count)
    return count

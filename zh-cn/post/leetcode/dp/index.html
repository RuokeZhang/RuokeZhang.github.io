def kadane(arr):
    res=arr[0]
    cur=arr[0]
    for i in range(1,len(arr)):
        cur=max(cur+arr[i], arr[i])
        res=max(res, cur)
    return res

def knapsack_exact_max(items, C):
    NEG_INF = -10**18
    dp = [NEG_INF] * (C + 1)
    dp[0] = 0

    for w, v in items:
        for c in range(C, w - 1, -1):
            if dp[c - w] != NEG_INF:
                dp[c] = max(dp[c], dp[c - w] + v)

    return dp[C]  # 如果仍是 NEG_INF，说明恰好凑不出

def coinChange(self, coins: List[int], amount: int) -> int:
    #dfs(i, c)表示遍历到第 i个硬币的时候，当前总价值为 c，所需要的最少硬币数
    @cache
    def dfs(i, c):
        if i<0:
            return 0 if c==0 else inf #容量需恰好为 amount
        if coins[i]>c:
            return dfs(i-1, c)
        return min(dfs(i-1, c), dfs(i, c-coins[i])+1)
    res=dfs(len(coins)-1, amount)
    return res if res<inf else -1

def coinChange(self, coins: List[int], amount: int) -> int:
    n=len(coins)
    dp=[[inf]*(amount+1) for _ in range(n+1)]
    dp[0][0]=0
    for i, x in enumerate(coins):
        for j in range(amount+1):
            if x<=j:
                dp[i+1][j]=min(dp[i+1][j-x]+1, dp[i][j])            
            else:
                dp[i+1][j]=dp[i][j]
    return dp[n][amount] if dp[n][amount]<inf else -1

dfs(i, j)=dfs(i-1, j)+1

s: ct, t: cat

如果此时对S执行添加操作，相当于

def maxScore(self, a: List[int], b: List[int]) -> int:
    #dp[i][j]=max(dp[i][j-1], dp[i-1][j-1]+a[i-1]*b[j-1])
    m=len(a)
    n=len(b)
    dp=[[-inf]*(n+1) for _ in range(m+1)]
    dp[0]=[0]*(n+1)
    for i in range(m):
        for j in range(n):
            dp[i+1][j+1]=max(dp[i+1][j], dp[i][j]+a[i]*b[j])
    return dp[m][n]
    #dp[1][2]=max(dp[1][1], dp[0][1]+a[0]*b[1])

初始化dp[0]=[0]*(n+1)，因为考虑到数组 a遍历到0（还没开始遍历的时候），不管b数组遍历到哪里，初始的乘积和为 0. 也可思考循环赋值过程，如最后一行的注释。

  • 优化方案：f[mask] = min(f[子集1] + f[子集2])

树形 DP

968. 监控二叉树

对于一个节点，有三个方法可以被监控

  • 自己装摄像头 （蓝）
  • 自己没装，parent 装摄像头 （黄）
  • 自己没装，child 装摄像头 （红）

考虑节点为蓝/黄/红

  • 当节点为蓝时：子节点可以为蓝/黄/红，则答案为：min(左蓝，左黄，左红)+min(右蓝，右黄，右红)+1
  • 黄：子节点不可能是黄色，可以是蓝红。则答案为：min(左蓝，左红)+min(右蓝，右红)
  • 红：子节点不可能是黄色，并且至少有一个儿子是蓝色。则答案为：min(左蓝+右红，右蓝+左红，左蓝+右蓝)

题目中的祖先（根节点）应当为蓝/红，dfs 函数返回三种情况所需的最少摄像头数量

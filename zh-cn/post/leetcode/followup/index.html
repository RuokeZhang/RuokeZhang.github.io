nums = [10,9,2,5,3,7,101,18]
import bisect
def sol(nums):
    g=[]
    g_idx=[]
    n=len(nums)
    parent=[-1]*n #记录nums[i]在最终答案里的上一个数字的index
    for i, x in enumerate(nums):
        idx=bisect.bisect_left(g, x)
        if idx>0:
            # its has a preceding number
            parent[i]=g_idx[idx-1]
        if idx<len(g):
            g[idx]=x
            g_idx[idx]=i
        else:
            g.append(x)
            g_idx.append(i)
    cur_idx=g_idx[-1]
    res=[]
    while cur_idx!=-1:
        res.append(nums[cur_idx])
        cur_idx=parent[cur_idx]
    return res[::-1]

def sol(nums):
    nums=[-inf]+nums+[-inf]
    n=len(nums)
    pre_nxt={}
    for i in range(1,n-1):
        pre_nxt[i]=[i-1,i+1]
    pq=[]
    for key,(pre,nxt) in pre_nxt.items():
        if nums[key]>nums[pre] and nums[key]>nums[nxt]:
            heapq.heappush(pq, (nums[key],key))
    res=[]
    #得到最小的peak
    while len(res)<n-2:
        min_peak,min_peak_idx=heapq.heappop(pq)
        res.append(min_peak)
        pre,nxt=pre_nxt[min_peak_idx]
        del pre_nxt[min_peak_idx]
        if pre in pre_nxt:
            pre_nxt[pre][1]=nxt
        if nxt in pre_nxt:
            pre_nxt[nxt][0]=pre
        #pre和next有可能变成peak
        for i in [pre,nxt]:
            if i in pre_nxt and (nums[i]>max([nums[idx] for idx in pre_nxt[i]])):
                heapq.heappush(pq, (nums[i],i))
    return res

def sol(root):
    res=-inf
    def dfs(root):
        nonlocal res
        if not root:
            return -inf
        if root.left==root.right:
            return root.val
        l=dfs(root.left)
        r=dfs(root.right)
        if not (l==-inf or r==-inf):
            res=max(res, root.val+l+r)
        return max(l,r)+root.val
    dfs(root)
    return res if res>-inf else 0

def findArray(n):
    state=[0]*n
    res=[]
    path=[]
    def dfs(step):
        if step==n*2:
            res.append(path.copy())
            return
        #遍历每个订单
        for i in range(n):
            if state[i]==0:
                #可以pickup
                path.append(f'P{i}')
                state[i]=1
                dfs(step+1)
                path.pop()
                state[i]=0
            elif state[i]==1:
                #可以deliver
                path.append(f'D{i}')
                state[i]=2
                dfs(step+1)
                path.pop()
                state[i]=1
    dfs(0)
    return res

def findSubArray(events):
    left=0
    unfinished=0
    res=0
    state=defaultdict(int)
    for right, event in enumerate(events):
        if event[0]=='P':
            event_id=event[1:]
            if state[event_id]!=0:
                #不valid
                state.clear()
                #但是现在的pickup可以作为下一个窗口的开端
                left=right
                state[event_id]=1
                unfinished+=1
                
            else:
                state[event_id]=1
                unfinished+=1
        else:
            event_id=event[1:]
            if state[event_id]!=1:
                #不valid
                state.clear()
                unfinished=0
                left=right+1
            else:
                state[event_id]=2
                unfinished-=1
                if unfinished==0:
                    res=max(res, right-left+1)
    return res

class Solution:
    def numDecodings(self, s: str) -> int:
        def dfs(start):
            #output all decodig representation of s[:start]
     
            if start==0:
                return 1, []
            res_num, res_path=0, []
            num=int(s[start-1:start])
            if num>0:
                c=chr(num+ord('A')-1)
                one_ways, one_path=dfs(start-1)
                if one_ways>0:
                    res_num+=one_ways
                    if len(one_path)==0:
                        res_path.append([c])
                    else:
                        for path in one_path:
                            path.append(c)
                            res_path.append(path.copy())
            if start>1:
                if s[start-2]!='0':
                    num=int(s[start-2:start])
                    if 1<=num<=26:
                        c=chr(num+ord('A')-1)
                        two_ways, two_path=dfs(start-2)
                        if two_ways>0:
                            res_num+=two_ways
                            if len(two_path)==0:
                                res_path.append([c])
                            else:                                   
                                for path in two_path:
                                    path.append(c)
                                    res_path.append(path.copy())
            return res_num, res_path
        res_num, path=dfs(len(s))
        print(path)
        return res_num
